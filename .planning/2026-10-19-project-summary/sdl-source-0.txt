bool Game::init(const char* title, int xpos, int ypos, int width, int height, int flags) {

	if (SDL_Init(SDL_INIT_EVERYTHING) == 0) {
		std::cout << "SDL initialized successfully" << std::endl;
		m_pWindow = SDL_CreateWindow(title, xpos, ypos, width, height, flags);

		if (m_pWindow != nullptr) {

			std::cout << "Create window successfully" << std::endl;
			m_pRenderer = SDL_CreateRenderer(m_pWindow, -1, 0);
			if (m_pRenderer == nullptr) {
				std::cout << "Renderer initialization failed for " << SDL_GetError() << std::endl;
				return false;
			}
			std::cout << "Renderer initialization successfully" << std::endl;
		}
		else {
			std::cout << "window initialization failed for " << SDL_GetError() << std::endl;
			return false;
		}

		// 注册工厂能够创建的对象类型

		TheGameObjectFactory::instance()->registerType("MenuButton",
			new MenuButtonCreator());
		TheGameObjectFactory::instance()->registerType("Player",
			new PlayerCreator());
		TheGameObjectFactory::instance()->registerType("Enemy",
			new EnemyCreator());
		TheGameObjectFactory::instance()->registerType("AnimatedGraphic",
			new AnimatedGraphicCreator());

		// 初始化进入 MenuState
		m_pGameStateMachine = new GameStateMachine();
		m_pGameStateMachine->pushState(new MainMenuState());

		// 初始化joystick
		TheInputHandler::Instance()->initialiseJoysticks();

		m_gameHeight = height;
		m_gameWidth = width;
	}
	// window and render initialized successfully
	std::cout << "Game initialized successfully" << std::endl;


	return true;
}

void Game::render() {
	SDL_RenderClear(m_pRenderer);

	// Render the current game state
	if (m_pGameStateMachine == nullptr) {
		std::cerr << "GameStateMachine is not initialized!" << std::endl;
		return;
	}
	m_pGameStateMachine->render();

	SDL_RenderPresent(m_pRenderer);
}

单条更新链
	Game::update()
	-> GameStateMachine::update()
		-> PlayState::update()
			-> 遍历 PlayState 的 m_gameObjects 调用各自 update()

	+------------+      +---------------------+      +--------------------+
	|   Game     | ---> |   GameStateMachine  | ---> |   Current GameState|
	|            |      |                     |      | (e.g., PlayState)  |
	|            |      |                     |      |                    |
	| update()   | ---> | update()            | ---> | update()           |
	|            |      |                     |      | - update GameObject|
	+------------+      +---------------------+      +--------------------+
*/

	m_pGameStateMachine->update();
}

bool GameObjectFactory::registerType(std::string typeID, BaseCreator* pCreator)
{
    std::map<std::string, BaseCreator*>::iterator it =
        m_creators.find(typeID);

	// Check if the typeID already exists
    if (it != m_creators.end())
    {
        delete pCreator;
        return false;
    }
    
    m_creators[typeID] = pCreator;

    return true;
}

GameObject* GameObjectFactory::create(std::string typeID)
{
    std::map<std::string, BaseCreator*>::iterator it =
        m_creators.find(typeID);

    if (it == m_creators.end())
    {
        std::cout << "could not find type: " << typeID << "\n";
        return NULL;
    }

	// Create the game object using the creator associated with the typeID
    BaseCreator* pCreator = (*it).second;
    return pCreator->createGameObject();
}

GameObjectFactory* GameObjectFactory::instance()
{
    static GameObjectFactory instance;
    return &instance;
}

}

void GameOverState::update()
{
	// Update logic for the game over state
	for (auto& gameObject : m_gameObjects) {
		gameObject->update(); // Update each game object
	}
	// Check for input to restart or return to main menu
	if (TheInputHandler::Instance()->isKeyDown(SDL_SCANCODE_RETURN)) {
		s_restartPlay(); // Restart the game
	} else if (TheInputHandler::Instance()->isKeyDown(SDL_SCANCODE_ESCAPE)) {
		s_gameOverToMain(); // Return to main menu
	}
}

void GameOverState::render()
{
	// Render logic for the game over state
	for (auto& gameObject : m_gameObjects) {
		gameObject->draw(); // Draw each game object
	}
}

bool GameOverState::onEnter()
{
	std::cout << "Entering GameOver State " << std::endl;
	StateParser parser;
	parser.parseState("assets/test.xml", s_gameOverID, &m_gameObjects, &m_textureIDList);

	m_callbacks.push_back(0);
	m_callbacks.push_back(s_gameOverToMain);
	m_callbacks.push_back(s_restartPlay);
	setCallbacks(m_callbacks);

	return true;
}

bool GameOverState::onExit()
{
	// 清理游戏对象
	for (auto& gameObject : m_gameObjects) {
		gameObject->clean();
	}
	m_gameObjects.clear(); // 清空游戏对象列表

	for (int i = 0; i < m_textureIDList.size(); i++)
	{
		TheTextureManager::Instance()
			->clearFromTextureMap(m_textureIDList[i]);
	}

	std::cout << "Exiting GameOver State" << std::endl;
	return true;
}

void GameOverState::setCallbacks(const std::vector<Callback>& callbacks)
{
	// 遍历所有 GameObject，找出是 MenuButton 类型的，
	// 然后根据 callbackID 从 callbacks 向量中找到对应函数，并通过 setCallback() 绑定给这个按钮。
	for (int i = 0; i < m_gameObjects.size(); i++)
	{
		if (dynamic_cast<MenuButton*>(m_gameObjects[i]))
		{
			MenuButton* pButton =
				dynamic_cast<MenuButton*>(m_gameObjects[i]);
			pButton->setCallback(callbacks[pButton->getCallbackID()]); // 通过 setCallback() 绑定给这个按钮。
		}
	}
}

#include "GameStateMachine.h"
#include <iostream>

// 入栈但不删除栈顶
void GameStateMachine::pushState(GameState* pState) {
	m_gameStates.push_back(pState);
	// initialize the new state
	m_gameStates.back()->onEnter();
}

// 删除栈顶
void GameStateMachine::popState() {
	if (!m_gameStates.empty()) {
		if (m_gameStates.back()->onExit()) {
			/*
				m_gameStates 是一个 std::vector<GameState*>，也就是说它存的是指针；
				如果只调用 pop_back()，指针从 vector 中移除了，但内存没释放，导致内存泄漏；
				如果只调用 delete，而不 pop_back()，那么这个“野指针”还留在容器里，之后访问会导致崩溃；
			*/
			delete m_gameStates.back();
			m_gameStates.pop_back();
		}
		else {
			// Handle failure to exit state
			std::cerr << "Failed to exit state: " << m_gameStates.back()->getStateId() << std::endl;
			return;
		}
	}
}

// 切换状态，先删除栈顶，再入栈新状态
void GameStateMachine::changeState(GameState* pState) {

	if (!m_gameStates.empty()) {

		// Check if the new state is the same as the current state
		if ( pState->getStateId() == m_gameStates.back()->getStateId() ) {
			return;
		}

		if (m_gameStates.back()->onExit()) { // 退出状态
			delete m_gameStates.back();
			m_gameStates.pop_back();
		}
		else {
			// Handle failure to exit state
			std::cerr << "Failed to exit state: " << m_gameStates.back()->getStateId() << std::endl;
			return;
		}
	}
	m_gameStates.push_back(pState);
	// Initialize the new state
	m_gameStates.back()->onEnter(); // 进入状态
}


void GameStateMachine::update() {
	if (!m_gameStates.empty()) {
		m_gameStates.back()->update();
	}
}

void GameStateMachine::render() {
	if (!m_gameStates.empty()) {
		m_gameStates.back()->render();
	}
}
