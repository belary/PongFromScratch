FILE Movement/Vector2D.h
#pragma once
#include <corecrt_math.h>
class Vector2D {
public:
	Vector2D(float x = 0, float y = 0) : m_x(x), m_y(y) {}
	float getX() const { return m_x; }
	float getY() const { return m_y; }

	void setX(float x) { m_x = x; }
	void setY(float y) { m_y = y; }

	float length() const { return sqrt(m_x * m_x + m_y * m_y); }

	// 参数 const Vector2D& v2 中的 const：表示参数 v2 是一个常量引用，函数内部不能修改这个参数的值
	// 函数末尾的 const：表示这个成员函数不会修改当前对象（this）的成员变量
	Vector2D operator+(const Vector2D& v2) const {
		// 栈上创建：Vector2D v(1, 2); 当函数返回时，这个临时对象会被复制到调用者的位置
		/*
			为什么使用栈而不是堆：
			性能更好：栈上分配内存比堆上分配快得多
			自动管理：栈上的对象会在作用域结束时自动销毁
			避免内存泄漏：不需要手动 delete
			符合 C++ 的值语义：向量运算通常返回新的值而不是指针
		*/
		return Vector2D(m_x + v2.getX(), m_y + v2.getY());
	}

	// 如果没有返回引用，
	// 1.链式操作就无法实现
	// 2.每次操作都会创建新的临时对象，性能会变差
	Vector2D& operator+=(const Vector2D &v2) {
		m_x += v2.m_x;
		m_y += v2.m_y;
		return *this;	
	}

	Vector2D operator-(const Vector2D& v2) const {
		return Vector2D(m_x - v2.m_x, m_y - v2.m_y);
	}

	Vector2D& operator-=(const Vector2D &v2) {
		m_x -= v2.m_x;
		m_y -= v2.m_y;
		return *this;
	}

	Vector2D operator*(float scalar) {
		return Vector2D(m_x * scalar, m_y * scalar);	
	}

	Vector2D& operator*=(float scalar) {
		m_x *= scalar;
		m_y *= scalar;
		return *this;
	}

	Vector2D operator/(float scalar) {
		return Vector2D(m_x / scalar, m_y / scalar);
	}

	Vector2D& operator/=(float scalar) {
		m_x /= scalar;
		m_y /= scalar;
		return *this;
	}
	
	void normalize() {
		float len = length();
		if (len > 0 ) {
			// 等价于  
			// m_x = m_x * (1/len)
			// m_y = m_y * (1/len)
			(*this) *= 1 / len;
		}
	}
	
	
private:
	float m_x, m_y;
};

bool Game::init(const char* title, int xpos, int ypos, int width, int height, int flags) {

	if (SDL_Init(SDL_INIT_EVERYTHING) == 0) {
		std::cout << "SDL initialized successfully" << std::endl;
		m_pWindow = SDL_CreateWindow(title, xpos, ypos, width, height, flags);

		if (m_pWindow != nullptr) {

			std::cout << "Create window successfully" << std::endl;
			m_pRenderer = SDL_CreateRenderer(m_pWindow, -1, 0);
			if (m_pRenderer == nullptr) {
				std::cout << "Renderer initialization failed for " << SDL_GetError() << std::endl;
				return false;
			}
			std::cout << "Renderer initialization successfully" << std::endl;
		}
		else {
			std::cout << "window initialization failed for " << SDL_GetError() << std::endl;
			return false;
		}

		// 注册工厂能够创建的对象类型

		TheGameObjectFactory::instance()->registerType("MenuButton",
			new MenuButtonCreator());
		TheGameObjectFactory::instance()->registerType("Player",
			new PlayerCreator());
		TheGameObjectFactory::instance()->registerType("Enemy",
			new EnemyCreator());
		TheGameObjectFactory::instance()->registerType("AnimatedGraphic",
			new AnimatedGraphicCreator());

		// 初始化进入 MenuState
		m_pGameStateMachine = new GameStateMachine();
		m_pGameStateMachine->pushState(new MainMenuState());

		// 初始化joystick
		TheInputHandler::Instance()->initialiseJoysticks();

		m_gameHeight = height;
		m_gameWidth = width;
	}
	// window and render initialized successfully
	std::cout << "Game initialized successfully" << std::endl;


	return true;
}

void Game::render() {
	SDL_RenderClear(m_pRenderer);

	// Render the current game state
	if (m_pGameStateMachine == nullptr) {
		std::cerr << "GameStateMachine is not initialized!" << std::endl;
		return;
	}
	m_pGameStateMachine->render();

	SDL_RenderPresent(m_pRenderer);
}

void Game::clean() {
	std::cout << "Cleaning game" << std::endl;
	SDL_DestroyWindow(m_pWindow);
	SDL_DestroyRenderer(m_pRenderer);
	TheInputHandler::Instance()->clean();
	SDL_Quit();
}


void Game::handleEvents() {
	TheInputHandler::Instance()->update();

	//if (TheInputHandler::Instance()->isKeyDown(SDL_SCANCODE_ESCAPE)) {
	//	m_bRunning = false;
	//}

	if (TheInputHandler::Instance()->isKeyDown(SDL_SCANCODE_RETURN)) {
		// 切换到 PlayState
		m_pGameStateMachine->changeState(new PlayState());
	}

}

bool Game::running() {
	return m_bRunning;
}

void Game::setRunning(bool running) {
	m_bRunning = running;
}

/*
单条更新链
	Game::update()
	-> GameStateMachine::update()
		-> PlayState::update()
			-> 遍历 PlayState 的 m_gameObjects 调用各自 update()

	+------------+      +---------------------+      +--------------------+
	|   Game     | ---> |   GameStateMachine  | ---> |   Current GameState|
	|            |      |                     |      | (e.g., PlayState)  |
	|            |      |                     |      |                    |
	| update()   | ---> | update()            | ---> | update()           |
	|            |      |                     |      | - update GameObject|
	+------------+      +---------------------+      +--------------------+
*/
void Game::update() {

	m_pGameStateMachine->update();
}

void Game::quit() {
	m_bRunning = false;
}

#include "TextureManager.h"

bool TextureManager::load(std::string fileName, std::string id, SDL_Renderer* pRenderer)
{
	SDL_Surface* pTempSurface = IMG_Load(fileName.c_str());
	if (pTempSurface == nullptr) {
		std::cout << "SDL_Image Error: " << IMG_GetError() << std::endl;
		return false;
	}
	
	SDL_Texture* pTexture = SDL_CreateTextureFromSurface(pRenderer, pTempSurface);
	SDL_FreeSurface(pTempSurface);

	if (pTexture == nullptr) {
		std::cout << "SDL_CreateTexture Error: " << SDL_GetError() << std::endl;
		return false;
	}
	m_textureMap[id] = pTexture;
	std::cout << "Texture loaded: " << id << std::endl;
    return true;
}

void TextureManager::draw(std::string id, int x, int y, int width, int height, SDL_Renderer* pRenderer, SDL_RendererFlip flip)
{
	SDL_Rect srcRect, destRect;
	srcRect.x = 0;
	srcRect.y = 0;
	srcRect.w = destRect.w = width;
	srcRect.h = destRect.h = height;

	destRect.x = x;
	destRect.y = y;
	if (SDL_RenderCopyEx(pRenderer, m_textureMap[id], &srcRect, &destRect, 0, nullptr, flip) < 0) {
		std::cout << "DRAW SDL_RenderCopyEx Error: " << SDL_GetError() << std::endl;
	}

}

void TextureManager::drawFrame(std::string id, int x, int y, int width, int height, int currentRow, int currentFrame, SDL_Renderer* pRenderer, SDL_RendererFlip flip)
{
	SDL_Rect srcRect, destRect;
	srcRect.x = currentFrame * width;
	srcRect.y = (currentRow - 1 ) * height;
	srcRect.w = destRect.w = width;
	srcRect.h = destRect.h = height;
	destRect.x = x;
	destRect.y = y;
	if (SDL_RenderCopyEx(pRenderer, m_textureMap[id], &srcRect, &destRect, 0, nullptr, flip) < 0) {
		std::cout << "DRAWFRAME SDL_RenderCopyEx Error: " << SDL_GetError() << std::endl;
	}
}

void TextureManager::drawTile(std::string id, int margin, int spacing, int x, int y, int width, int height, 
	int currentRow, int currentFrame, SDL_Renderer* pRenderer)
{
	SDL_Rect srcRect, destRect;
	srcRect.x = margin + (spacing + width ) * currentFrame; // 这里是列
	srcRect.y = margin + (spacing + height) * currentRow;   // 这里是行
	srcRect.w = destRect.w = width;
	srcRect.h = destRect.h = height;
	destRect.x = x;
	destRect.y = y;
	if (SDL_RenderCopyEx(pRenderer, m_textureMap[id], &srcRect, &destRect, 0, nullptr, SDL_FLIP_NONE) < 0) {
		std::cout << "DRAWFRAME SDL_RenderCopyEx Error: " << SDL_GetError() << std::endl;
	}
}

TextureManager* TextureManager::Instance()
{
	static TextureManager instance;
	return &instance;
}

void TextureManager::clearFromTextureMap(std::string id)
{
	auto it = m_textureMap.find(id);
	if (it != m_textureMap.end()) {
		SDL_DestroyTexture(it->second);
		m_textureMap.erase(it);
		std::cout << "Texture cleared: " << id << std::endl;
	}
	else {
		std::cout << "Texture not found: " << id << std::endl;
	}
}
