FILE AlienAttack/Vector2D.h
#pragma once
#include <cmath>
class Vector2D {
public:
	Vector2D(float x = 0, float y = 0) : m_x(x), m_y(y) {}
	float getX() const { return m_x; }
	float getY() const { return m_y; }

	void setX(float x) { m_x = x; }
	void setY(float y) { m_y = y; }

	float length() const { return sqrt(m_x * m_x + m_y * m_y); }

	// 参数 const Vector2D& v2 中的 const：表示参数 v2 是一个常量引用，函数内部不能修改这个参数的值
	// 函数末尾的 const：表示这个成员函数不会修改当前对象（this）的成员变量
	Vector2D operator+(const Vector2D& v2) const {
		// 栈上创建：Vector2D v(1, 2); 当函数返回时，这个临时对象会被复制到调用者的位置
		/*
			为什么使用栈而不是堆：
			性能更好：栈上分配内存比堆上分配快得多
			自动管理：栈上的对象会在作用域结束时自动销毁
			避免内存泄漏：不需要手动 delete
			符合 C++ 的值语义：向量运算通常返回新的值而不是指针
		*/
		return Vector2D(m_x + v2.getX(), m_y + v2.getY());
	}

	// 如果没有返回引用，
	// 1.链式操作就无法实现
	// 2.每次操作都会创建新的临时对象，性能会变差
	Vector2D& operator+=(const Vector2D &v2) {
		m_x += v2.m_x;
		m_y += v2.m_y;
		return *this;	
	}

	Vector2D operator-(const Vector2D& v2) const {
		return Vector2D(m_x - v2.m_x, m_y - v2.m_y);
	}

	Vector2D& operator-=(const Vector2D &v2) {
		m_x -= v2.m_x;
		m_y -= v2.m_y;
		return *this;
	}

	Vector2D operator*(float scalar) {
		return Vector2D(m_x * scalar, m_y * scalar);	
	}

	Vector2D& operator*=(float scalar) {
		m_x *= scalar;
		m_y *= scalar;
		return *this;
	}

	Vector2D operator/(float scalar) {
		return Vector2D(m_x / scalar, m_y / scalar);
	}

	Vector2D& operator/=(float scalar) {
		m_x /= scalar;
		m_y /= scalar;
		return *this;
	}
	
	void normalize() {
		float len = length();
		if (len > 0 ) {
			// 等价于  
			// m_x = m_x * (1/len)
			// m_y = m_y * (1/len)
			(*this) *= 1 / len;
		}
	}
	
private:
	float m_x, m_y;
};

			return false;
		}

		// 注册工厂能够创建的对象类型

		TheGameObjectFactory::instance()->registerType("MenuButton",
			new MenuButtonCreator());
		TheGameObjectFactory::instance()->registerType("Player",

		TheGameObjectFactory::instance()->registerType("Enemy",

		TheGameObjectFactory::instance()->registerType("AnimatedGraphic",
			new AnimatedGraphicCreator());

		m_pGameStateMachine = new GameStateMachine();
		m_pGameStateMachine->pushState(new MainMenuState());

		// 初始化joystick
		TheInputHandler::Instance()->initialiseJoysticks();

	std::cout << "Game initialized successfully" << std::endl;


	return true;

	m_pGameStateMachine->render();

	SDL_RenderPresent(m_pRenderer);
}

FILE DataDriveDesign/GameStateMachine.cpp
#include "GameStateMachine.h"
#include <iostream>

// 入栈但不删除栈顶
void GameStateMachine::pushState(GameState* pState) {
	m_gameStates.push_back(pState);
	// initialize the new state
	m_gameStates.back()->onEnter();
}

// 删除栈顶
void GameStateMachine::popState() {
	if (!m_gameStates.empty()) {
		if (m_gameStates.back()->onExit()) {
			/*
				m_gameStates 是一个 std::vector<GameState*>，也就是说它存的是指针；
				如果只调用 pop_back()，指针从 vector 中移除了，但内存没释放，导致内存泄漏；
				如果只调用 delete，而不 pop_back()，那么这个“野指针”还留在容器里，之后访问会导致崩溃；
			*/
			delete m_gameStates.back();
			m_gameStates.pop_back();
		}
		else {
			// Handle failure to exit state
			std::cerr << "Failed to exit state: " << m_gameStates.back()->getStateId() << std::endl;
			return;
		}
	}
}

// 切换状态，先删除栈顶，再入栈新状态
void GameStateMachine::changeState(GameState* pState) {

	if (!m_gameStates.empty()) {

		// Check if the new state is the same as the current state
		if ( pState->getStateId() == m_gameStates.back()->getStateId() ) {
			return;
		}

		if (m_gameStates.back()->onExit()) {
			delete m_gameStates.back();
			m_gameStates.pop_back();
		}
		else {
			// Handle failure to exit state
			std::cerr << "Failed to exit state: " << m_gameStates.back()->getStateId() << std::endl;
			return;
		}
	}
	m_gameStates.push_back(pState);
	// Initialize the new state
	m_gameStates.back()->onEnter();
}


void GameStateMachine::update() {
	if (!m_gameStates.empty()) {
		m_gameStates.back()->update();
	}
}

void GameStateMachine::render() {
	if (!m_gameStates.empty()) {
		m_gameStates.back()->render();
	}
}

	std::cout << "onExit: Clearing " << m_gameObjects.size() << " objects" << std::endl;

	// 清理游戏对象
	for (auto& gameObject : m_gameObjects) {
		gameObject->clean();

	for (int i = 0; i < m_textureIDList.size(); i++)
	{
		TheTextureManager::Instance()->clearFromTextureMap(m_textureIDList[i]);
	}

	std::cout << "Exiting Play State" << std::endl;
	return true;
}

bool PlayState::checkCollision(SDLGameObject* p1, SDLGameObject* p2) {

	float leftA, leftB;
	float rightA, rightB;
	float topA, topB;
	float bottomA, bottomB;

	leftA = p1->getPosition().getX();
	rightA = p1->getPosition().getX() + p1->getWidth();
	topA = p1->getPosition().getY();
	bottomA = p1->getPosition().getY() + p1->getHeight();


	leftB = p2->getPosition().getX();
	rightB = p2->getPosition().getX() + p2->getWidth();
	topB = p2->getPosition().getY();
	bottomB = p2->getPosition().getY() + p2->getHeight();

	if (bottomA <= topB) {
		return false; // A is above B
	}
	if (topA >= bottomB) {
		return false; // A is below B
	}

	if (rightA <= leftB) {
		return false; // A is to the left of B
	}

	if (leftA >= rightB) {
		return false; // A is to the right of B
	}

	return true;
}
