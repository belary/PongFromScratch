/*
	一种成员函数指针类型。
	表示指向 MenuState 类里返回 void，不带参数的成员函数。
	void (MenuState::*)() 是 类型,描述「这是一个指向 MenuState 成员函数的指针类型
		
	使用了这个类型定义的变量 func, 声明了一个名为 func 的成员函数指针变量
	void (MenuState::*func)() 

	void(*)()				普通全局或 static 函数指针	void func()
	void(MenuState::*)()	成员函数指针 (需要对象调用)	MenuState::func()
*/


typedef void(*MenuCallback)(); //使用 typedef 给一个复杂的函数指针类型起了一个更简洁的名字 MenuCallback。

class MenuButton : public SDLGameObject {
public:
	MenuButton(const LoaderParams* pParams, MenuCallback callback);
	virtual void draw();
	virtual void update();
	virtual void clean();
private:
	enum button_state{
		MOUSE_OUT = 0,
		MOUSE_OVER = 1,
		CLICKED = 2
	};

	// 声明了一个成员变量 m_callback, 类型是刚刚通过 typedef 定义的 MenuCallback
	// 也就是一个指向 void() 函数的指针。
	MenuCallback m_callback; // 回调函数指针，用于按钮点击时调用

	bool m_bReleased; // 用于判断按钮是否被释放
};

/*
 *	在 onEnter 中创建对象，是为了让每个状态只关心并管理它自己需要的对象和资源。
 *	这样使 Game 类保持干净，FSM 切换时只需调用 onEnter / onExit，就能自动加载和清理资源。
 *	未来扩展其他状态（如 OptionsState、WinState）也不会需要修改 Game 或 MenuState。
 * 
 * 
 ┌───────────────────────┐
 │       Game            │
 │ (窗口/渲染器/入口点)     │
 └─────────┬─────────────┘
           │
           ▼
 ┌───────────────────────┐
 │  GameStateMachine     │
 │ (Finite State Machine)│
 └─────────┬─────────────┘
           │
     manages current
           ▼
 ┌───────────────────────┐
 │     GameState         │
 │  (抽象基类/接口)        │
 └─────────┬─────────────┘
           │
    ┌──────┴─────────────┬─────────────┐
    ▼                    ▼             ▼
┌───────────┐     ┌────────────┐  ┌────────────┐
│ MenuState │     │ PlayState  │  │ PauseState │
│ (onEnter) │     │ (onEnter)  │  │ (onEnter)  │
└─────┬─────┘     └─────┬──────┘  └─────┬──────┘
      │                 │               │
  owns GameObjects   owns GameObjects   owns GameObjects
 (按钮、标题等)       (玩家、敌人等)     (菜单按钮)

*/
bool MenuState::onEnter() {

    // 加载菜单图片资源
    if (!TheTextureManager::Instance()->load(
            "assets/button.png", 
            "playbutton", 
            TheGame::Instance()->getRenderer())) {
        std::cout << "Failed to load play button texture" << std::endl;
		return false;
    }

    if (!TheTextureManager::Instance()->load(
        "assets/exit.png",
        "exitbutton",
        TheGame::Instance()->getRenderer())) {
        std::cout << "Failed to load exit button texture" << std::endl;
        return false;
    }

	// 创建菜单按钮对象
	GameObject* playButton = new MenuButton(new LoaderParams(100, 100, 400, 100, "playbutton"), s_menuToPlay);
	GameObject* exitButton = new MenuButton(new LoaderParams(100, 300, 400, 100, "exitbutton"), s_exitFromMenu);

	// 加入状态管理器的游戏对象列表
	m_gameObjects.push_back(playButton);
	m_gameObjects.push_back(exitButton);


	std::cout << "Entering Menu State" << std::endl;
	return true;
}

bool MenuState::onExit() {
    for (auto& gameObject : m_gameObjects) {
        gameObject->clean(); // 清理每个游戏对象
	}
	m_gameObjects.clear(); // 清空游戏对象列表

	TheTextureManager::Instance()->clearFromTextureMap("playbutton");
	TheTextureManager::Instance()->clearFromTextureMap("exitbutton");
	std::cout << "Exiting Menu State" << std::endl;
	return true;
}

void MenuState::s_menuToPlay() {

	std::cout << "Switching to Play State" << std::endl;
    // 切换到 PlayState
    TheGame::Instance()->getStateMachine()->changeState(new PlayState());
}

void MenuState::s_exitFromMenu() {
	std::cout << "Exiting Game" << std::endl;
    // 退出游戏
    TheGame::Instance()->quit();
}

		m_velocity.setY(2);

const std::string PlayState::s_playID = "PLAY";

void PlayState::update() {


	// Check if the player wants to pause the game
	if (TheInputHandler::Instance()->isKeyDown(SDL_SCANCODE_ESCAPE)) {
		TheGame::Instance()->getStateMachine()->pushState(new PauseState()); // Push PauseState onto the stack
		return; // Exit early to avoid updating game objects while paused
	}
	// 
	// Update logic for the play state
	for (auto& gameObject : m_gameObjects) {
		gameObject->update();
	}

	if (m_gameObjects.size() > 1) {
		// Check for collisions between game objects
		if (checkCollision(dynamic_cast<SDLGameObject*>(m_gameObjects[0]), dynamic_cast<SDLGameObject*>(m_gameObjects[1]))) {
			std::cout << "Collision detected between object " << 0 << " and object " << 1 << std::endl;
			// Handle collision logic here
			TheGame::Instance()->getStateMachine()->changeState(new GameOverState()); // Change to GameOverState on collision
		}
	}
}

void PlayState::render() {
	// Render logic for the play state
	for (auto& gameObject : m_gameObjects) {
		gameObject->draw(); // Draw each game object
	}
}

bool PlayState::onEnter() {

bool PlayState::checkCollision(SDLGameObject* p1, SDLGameObject* p2) {

	int leftA, leftB;
	int rightA, rightB;
	int topA, topB;
	int bottomA, bottomB;

	leftA = p1->getPosition().getX();
	rightA = p1->getPosition().getX() + p1->getWidth();
	topA = p1->getPosition().getY();
	bottomA = p1->getPosition().getY() + p1->getHeight();


	leftB = p2->getPosition().getX();
	rightB = p2->getPosition().getX() + p2->getWidth();
	topB = p2->getPosition().getY();
	bottomB = p2->getPosition().getY() + p2->getHeight();

	if (bottomA <= topB) {
		return false; // A is above B
	}
	if (topA >= bottomB) {
		return false; // A is below B
	}

	if (rightA <= leftB) {
		return false; // A is to the left of B
	}

	if (leftA >= rightB) {
		return false; // A is to the right of B
	}

	return true;
}

FILE GameState/TextureManager.cpp
#include "TextureManager.h"

bool TextureManager::load(std::string fileName, std::string id, SDL_Renderer* pRenderer)
{
	SDL_Surface* pTempSurface = IMG_Load(fileName.c_str());
	if (pTempSurface == nullptr) {
		std::cout << "SDL_Image Error: " << IMG_GetError() << std::endl;
		return false;
	}
	
	SDL_Texture* pTexture = SDL_CreateTextureFromSurface(pRenderer, pTempSurface);
	SDL_FreeSurface(pTempSurface);

	if (pTexture == nullptr) {
		std::cout << "SDL_CreateTexture Error: " << SDL_GetError() << std::endl;
		return false;
	}
	m_textureMap[id] = pTexture;
	std::cout << "Texture loaded: " << id << std::endl;
    return true;
}

void TextureManager::draw(std::string id, int x, int y, int width, int height, SDL_Renderer* pRenderer, SDL_RendererFlip flip)
{
	SDL_Rect srcRect, destRect;
	srcRect.x = 0;
	srcRect.y = 0;
	srcRect.w = destRect.w = width;
	srcRect.h = destRect.h = height;

	destRect.x = x;
	destRect.y = y;
	if (SDL_RenderCopyEx(pRenderer, m_textureMap[id], &srcRect, &destRect, 0, nullptr, flip) < 0) {
		std::cout << "SDL_RenderCopyEx Error: " << SDL_GetError() << std::endl;
	}

}

void TextureManager::drawFrame(std::string id, int x, int y, int width, int height, int currentRow, int currentFrame, SDL_Renderer* pRenderer, SDL_RendererFlip flip)
{
	SDL_Rect srcRect, destRect;
	srcRect.x = currentFrame * width;
	srcRect.y = (currentRow - 1 ) * height;
	srcRect.w = destRect.w = width;
	srcRect.h = destRect.h = height;
	destRect.x = x;
	destRect.y = y;
	if (SDL_RenderCopyEx(pRenderer, m_textureMap[id], &srcRect, &destRect, 0, nullptr, flip) < 0) {
		std::cout << "SDL_RenderCopyEx Error: " << SDL_GetError() << std::endl;
	}
}

TextureManager* TextureManager::Instance()
{
	static TextureManager instance;
	return &instance;
}

void TextureManager::clearFromTextureMap(std::string id)
{
	auto it = m_textureMap.find(id);
	if (it != m_textureMap.end()) {
		SDL_DestroyTexture(it->second);
		m_textureMap.erase(it);
		std::cout << "Texture cleared: " << id << std::endl;
	} else {
		std::cout << "Texture not found: " << id << std::endl;
	}
}
